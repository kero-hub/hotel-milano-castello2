async function apiCall(payload) { var params = new URLSearchParams(); Object.keys(payload || {}).forEach(function (k) { var v = payload[k]; if (v !== undefined && v !== null && v !== '') params.set(k, String(v)); }); var url = SCRIPT_URL + (SCRIPT_URL.indexOf('?') >= 0 ? '&' : '?') + params.toString(); console.log('API URL:', url); var res = await fetch(url, { method: 'GET', redirect: 'follow', cache: 'no-cache', credentials: 'omit' }); var text = await res.text(); console.log('API RAW:', text); if (!text || text.trim().charAt(0) === '<') { throw new Error('Risposta non JSON dallo script'); } var json = JSON.parse(text); // tessera nuova if (json.needCreate) return json; if (json.ok === false) { throw new Error(json.error || 'Errore script'); } return json; } async function onCard(uid) { if (!uid || busy) return; uid = String(uid).toUpperCase().replace(/\s+/g, ''); document.getElementById('last-uid').textContent = uid; pendingUid = uid; busy = true; document.getElementById('create-box').classList.add('hidden'); try { var r = await apiCall({ action: 'timbra', uid: uid, operatore: loggedUserName || 'NFC' }); console.log('API JSON:', r); if (r.needCreate) { document.getElementById('new-uid').textContent = uid; document.getElementById('new-name').value = ''; document.getElementById('create-box').classList.remove('hidden'); toast('Tessera nuova: inserisci il nome', 'info'); return; } var t = r.timbratura; if (!t) { // se il foglio ha ricevuto il dato ma manca il campo, non bloccare toast('Timbratura OK (senza dettaglio risposta)', 'success'); loadDipendenti(); loadTurni(); return; } var tipo = String(t.tipo || '').toUpperCase(); var oraR = roundTo15(t.ora || ''); document.getElementById('live-msg').textContent = (tipo === 'IN' ? '🟢 INGRESSO' : '🔴 USCITA') + ' — ' + (t.nome || '') + (oraR !== '—' ? ' alle ' + oraR : ''); toast((t.nome || '') + ' → ' + tipo + ' ' + oraR, 'success'); loadDipendenti(); loadTurni(); } catch (err) { console.error(err); toast(err.message || 'Errore timbratura', 'error'); } finally { busy = false; } } async function creaETimbra() { var nome = document.getElementById('new-name').value.trim(); if (!nome) { toast('Inserisci il nome', 'error'); return; } if (!pendingUid) return; busy = true; try { var r = await apiCall({ action: 'timbra', uid: pendingUid, nome: nome, tipo: 'IN', operatore: loggedUserName || 'NFC' }); var t = r.timbratura; document.getElementById('create-box').classList.add('hidden'); if (t) { document.getElementById('live-msg').textContent = '🟢 INGRESSO — ' + t.nome + ' alle ' + roundTo15(t.ora || ''); toast('Creato ' + t.nome + ' + IN', 'success'); } else { toast('Dipendente salvato', 'success'); } loadDipendenti(); loadTurni(); } catch (err) { console.error(err); toast(err.message || 'Errore', 'error'); } finally { busy = false; } }
